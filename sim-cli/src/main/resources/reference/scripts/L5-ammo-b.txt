if (municion() == 0 || calor() >= 70) {
    recargar();
} else {
    Enemigo o = null;
    for (Enemigo e : enemigos()) {
        if (o == null || e.distanciaAlNucleo() < o.distanciaAlNucleo()) {
            o = e;
        }
    }
    if (o != null) {
        apuntar(o);
        if ({PERFORANTE_SI}) usarMunicion(PERFORANTE); else usarMunicion(NORMAL);
        disparar();
    }
}
